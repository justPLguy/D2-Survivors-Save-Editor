import base64
import json
import zlib
from pathlib import Path

# D2 Survivors Save Editor — Add Gold (No GUI)
# Created by Just a Polish Guy
# Unofficial community-made tool. Always back up your save.

SAVE_CODE = """
PASTE YOUR SAVE CODE HERE
""".strip()


def main():
    print("=" * 48)
    print(" D2 SURVIVORS - ADD GOLD EDITOR")
    print(" Created by Just a Polish Guy")
    print("=" * 48)
    print("Unofficial community tool — use at your own risk.")
    print()

    try:
        if not SAVE_CODE.startswith("D2S1:"):
            raise ValueError(
                "Invalid save-code prefix. Replace SAVE_CODE with your full D2S1: save code."
            )

        encoded = SAVE_CODE[len("D2S1:"):]
        encoded += "=" * (-len(encoded) % 4)

        raw = base64.urlsafe_b64decode(encoded)
        data = json.loads(zlib.decompress(raw))

        if not isinstance(data, dict):
            raise ValueError("Save data is not a JSON object.")

        if "gold" not in data:
            available_fields = ", ".join(data.keys())
            raise ValueError(
                f"Could not find the 'gold' field. Available fields: {available_fields}"
            )

        print("Save loaded successfully.")
        print("Current gold:", data["gold"])
        print("Save version:", data.get("version", "unknown"))

        new_gold = int(input("Enter the desired amount of gold: "))
        if new_gold < 0:
            raise ValueError("Gold cannot be negative.")

        data["gold"] = new_gold

        json_bytes = json.dumps(
            data,
            separators=(",", ":"),
            ensure_ascii=False
        ).encode("utf-8")

        new_save = "D2S1:" + base64.urlsafe_b64encode(
            zlib.compress(json_bytes)
        ).decode("ascii").rstrip("=")

        output_path = Path(__file__).resolve().with_name("new_save.txt")
        output_path.write_text(new_save, encoding="utf-8")

        print()
        print("The new save has been written to:")
        print(output_path)
        print("New gold amount:", new_gold)
        print()
        print("The original save has not been modified.")

    except Exception as error:
        print()
        print("ERROR:", error)
        print("The original save has not been modified.")

    print()
    print("Credits: Just a Polish Guy")
    print("GitHub: https://github.com/justPLguy/D2-Survivors-Save-Editor")
    input("\nPress Enter to exit...")


if __name__ == "__main__":
    main()
